def capture_stdout(&block)
  stdout, string = $stdout, StringIO.new
  $stdout = string

  yield

  string.tap(&:rewind).read
ensure
  $stdout = stdout
end

def gemfile
  gemfile = ENV['BUNDLE_GEMFILE']

  File.basename(gemfile, '.*') if gemfile
end
